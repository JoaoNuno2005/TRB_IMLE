      "severity": "low|medium|high",

      "affected_area_pct": 0.0
    }
  ],
  "shelf_fill_rate": 0.0,
  "products_detected": ["categorias visíveis"],
  "model_reasoning": "evidência visual observável usada antes da classificação final",
